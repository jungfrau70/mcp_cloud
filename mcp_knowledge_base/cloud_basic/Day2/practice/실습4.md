# 🏦 AWS 서버리스 투자 제안 심사 시스템 실습 가이드 (웹콘솔 + CLI)

## 📋 개요

**목적**: AWS 서버리스 서비스를 활용하여 투자 제안 심사 시스템을 구축하고 서버리스 아키텍처 경험
**범위**: 
- AWS Lambda 함수 개발 및 배포
- Amazon API Gateway를 통한 REST API 구축
- Amazon DynamoDB를 활용한 데이터 저장
- AWS Step Functions를 통한 워크플로우 자동화
- Amazon S3를 활용한 문서 저장
- Amazon SNS를 통한 알림 시스템
- Amazon CloudWatch를 통한 모니터링

## 🔧 4단계: Lambda 함수 개발

### 4.1 투자 제안 업로드 함수

#### 💻 CLI 방식
```bash
# 작업 디렉토리 생성
mkdir lambda-functions
cd lambda-functions

# 제안 업로드 Lambda 함수 생성
cat > proposal_upload.py << 'EOF'
import json
import boto3
import uuid
from datetime import datetime
import base64

dynamodb = boto3.resource('dynamodb')
s3 = boto3.client('s3')
table = dynamodb.Table('InvestmentProposals')

def lambda_handler(event, context):
    try:
        # 요청 본문 파싱
        body = json.loads(event['body'])
        
        # 제안 ID 생성
        proposal_id = str(uuid.uuid4())
        
        # 제안 데이터 준비
        proposal_data = {
            'proposal_id': proposal_id,
            'company_name': body.get('company_name'),
            'investment_amount': body.get('investment_amount'),
            'business_description': body.get('business_description'),
            'expected_return': body.get('expected_return'),
            'risk_level': body.get('risk_level'),
            'status': 'PENDING',
            'created_date': datetime.now().isoformat(),
            'updated_date': datetime.now().isoformat()
        }
        
        # DynamoDB에 저장
        table.put_item(Item=proposal_data)
        
        # S3에 문서 저장 (있는 경우)
        if 'document' in body:
            document_content = base64.b64decode(body['document'])
            s3.put_object(
                Bucket='investment-proposals-docs-' + datetime.now().strftime('%Y%m%d'),
                Key=f'documents/{proposal_id}.pdf',
                Body=document_content
            )
        
        return {
            'statusCode': 200,
            'headers': {
                'Content-Type': 'application/json',
                'Access-Control-Allow-Origin': '*'
            },
            'body': json.dumps({
                'message': 'Investment proposal uploaded successfully',
                'proposal_id': proposal_id
            })
        }
        
    except Exception as e:
        return {
            'statusCode': 500,
            'headers': {
                'Content-Type': 'application/json',
                'Access-Control-Allow-Origin': '*'
            },
            'body': json.dumps({
                'error': str(e)
            })
        }
EOF

# Lambda 함수 배포 패키지 생성
zip -r proposal_upload.zip proposal_upload.py

# Lambda 함수 생성
aws lambda create-function \
  --function-name proposal-upload \
  --runtime python3.9 \
  --role arn:aws:iam::$(aws sts get-caller-identity --query Account --output text):role/lambda-execution-role \
  --handler proposal_upload.lambda_handler \
  --zip-file fileb://proposal_upload.zip \
  --timeout 30 \
  --memory-size 256
```

### 4.2 제안 분석 함수

#### 💻 CLI 방식
```bash
# 제안 분석 Lambda 함수 생성
cat > proposal_analysis.py << 'EOF'
import json
import boto3
from datetime import datetime

dynamodb = boto3.resource('dynamodb')
table = dynamodb.Table('InvestmentProposals')

def lambda_handler(event, context):
    try:
        # 제안 ID 추출
        proposal_id = event['proposal_id']
        
        # DynamoDB에서 제안 데이터 조회
        response = table.get_item(Key={'proposal_id': proposal_id})
        proposal = response['Item']
        
        # 분석 로직
        analysis_result = analyze_proposal(proposal)
        
        # 분석 결과 저장
        table.update_item(
            Key={'proposal_id': proposal_id},
            UpdateExpression='SET analysis_result = :result, updated_date = :date',
            ExpressionAttributeValues={
                ':result': analysis_result,
                ':date': datetime.now().isoformat()
            }
        )
        
        return {
            'proposal_id': proposal_id,
            'analysis_result': analysis_result
        }
        
    except Exception as e:
        raise Exception(f"Analysis failed: {str(e)}")

def analyze_proposal(proposal):
    # 간단한 분석 로직
    investment_amount = float(proposal['investment_amount'])
    expected_return = float(proposal['expected_return'])
    risk_level = proposal['risk_level']
    
    # 위험도별 승인 기준
    risk_thresholds = {
        'LOW': 0.05,
        'MEDIUM': 0.10,
        'HIGH': 0.15
    }
    
    threshold = risk_thresholds.get(risk_level, 0.10)
    roi = expected_return / investment_amount
    
    if roi >= threshold:
        return {
            'status': 'APPROVED',
            'score': min(100, int(roi * 1000)),
            'reason': f'ROI {roi:.2%} meets threshold {threshold:.2%}'
        }
    else:
        return {
            'status': 'REJECTED',
            'score': max(0, int(roi * 1000)),
            'reason': f'ROI {roi:.2%} below threshold {threshold:.2%}'
        }
EOF

# Lambda 함수 배포 패키지 생성
zip -r proposal_analysis.zip proposal_analysis.py

# Lambda 함수 생성
aws lambda create-function \
  --function-name proposal-analysis \
  --runtime python3.9 \
  --role arn:aws:iam::$(aws sts get-caller-identity --query Account --output text):role/lambda-execution-role \
  --handler proposal_analysis.lambda_handler \
  --zip-file fileb://proposal_analysis.zip \
  --timeout 30 \
  --memory-size 256
```

### 4.3 알림 함수

#### 💻 CLI 방식
```bash
# 알림 Lambda 함수 생성
cat > notification.py << 'EOF'
import json
import boto3

sns = boto3.client('sns')

def lambda_handler(event, context):
    try:
        # 알림 메시지 생성
        proposal_id = event['proposal_id']
        analysis_result = event['analysis_result']
        
        message = f"""
Investment Proposal Review Complete
        
Proposal ID: {proposal_id}
Status: {analysis_result['status']}
Score: {analysis_result['score']}/100
Reason: {analysis_result['reason']}
        
Review completed at: {event.get('timestamp', 'N/A')}
        """
        
        # SNS 토픽으로 알림 발송
        response = sns.publish(
            TopicArn=event['topic_arn'],
            Subject=f'Investment Proposal Review - {analysis_result["status"]}',
            Message=message
        )
        
        return {
            'message': 'Notification sent successfully',
            'message_id': response['MessageId']
        }
        
    except Exception as e:
        raise Exception(f"Notification failed: {str(e)}")
EOF

# Lambda 함수 배포 패키지 생성
zip -r notification.zip notification.py

# Lambda 함수 생성
aws lambda create-function \
  --function-name notification \
  --runtime python3.9 \
  --role arn:aws:iam::$(aws sts get-caller-identity --query Account --output text):role/lambda-execution-role \
  --handler notification.lambda_handler \
  --zip-file fileb://notification.zip \
  --timeout 30 \
  --memory-size 256
```

## 🌐 5단계: API Gateway 설정

### 5.1 REST API 생성

#### 🌐 웹콘솔 방식
```markdown
1. API Gateway 콘솔 접속
2. "API 생성" 클릭
3. "REST API" 선택
4. API 이름: "Investment Proposals API"
5. 설명: "투자 제안 심사 시스템 API"
6. "API 생성" 클릭
```

#### 💻 CLI 방식
```bash
# REST API 생성
aws apigateway create-rest-api \
  --name "Investment Proposals API" \
  --description "투자 제안 심사 시스템 API"

# API ID 저장
API_ID=$(aws apigateway get-rest-apis --query 'items[?name==`Investment Proposals API`].id' --output text)

# 루트 리소스 ID 가져오기
ROOT_ID=$(aws apigateway get-resources --rest-api-id $API_ID --query 'items[?path==`/`].id' --output text)

# /proposals 리소스 생성
PROPOSALS_ID=$(aws apigateway create-resource \
  --rest-api-id $API_ID \
  --parent-id $ROOT_ID \
  --path-part "proposals" \
  --query 'id' --output text)

# POST 메서드 생성
aws apigateway put-method \
  --rest-api-id $API_ID \
  --resource-id $PROPOSALS_ID \
  --http-method POST \
  --authorization-type NONE

# Lambda 함수 연동
aws apigateway put-integration \
  --rest-api-id $API_ID \
  --resource-id $PROPOSALS_ID \
  --http-method POST \
  --type AWS_PROXY \
  --integration-http-method POST \
  --uri arn:aws:apigateway:ap-northeast-2:lambda:path/2015-03-31/functions/arn:aws:lambda:ap-northeast-2:$(aws sts get-caller-identity --query Account --output text):function:proposal-upload/invocations

# Lambda 함수에 API Gateway 권한 부여
aws lambda add-permission \
  --function-name proposal-upload \
  --statement-id apigateway-invoke \
  --action lambda:InvokeFunction \
  --principal apigateway.amazonaws.com \
  --source-arn "arn:aws:execute-api:ap-northeast-2:$(aws sts get-caller-identity --query Account --output text):$API_ID/*/*"
```

## 🔄 6단계: Step Functions 워크플로우

### 6.1 상태 머신 정의

#### 💻 CLI 방식
```bash
# Step Functions 상태 머신 정의
cat > state-machine-definition.json << 'EOF'
{
  "Comment": "Investment Proposal Review Workflow",
  "StartAt": "AnalyzeProposal",
  "States": {
    "AnalyzeProposal": {
      "Type": "Task",
      "Resource": "arn:aws:lambda:ap-northeast-2:ACCOUNT_ID:function:proposal-analysis",
      "Next": "CheckAnalysisResult",
      "Catch": [
        {
          "ErrorEquals": ["States.ALL"],
          "Next": "HandleError"
        }
      ]
    },
    "CheckAnalysisResult": {
      "Type": "Choice",
      "Choices": [
        {
          "Variable": "$.analysis_result.status",
          "StringEquals": "APPROVED",
          "Next": "SendApprovalNotification"
        },
        {
          "Variable": "$.analysis_result.status",
          "StringEquals": "REJECTED",
          "Next": "SendRejectionNotification"
        }
      ],
      "Default": "HandleError"
    },
    "SendApprovalNotification": {
      "Type": "Task",
      "Resource": "arn:aws:lambda:ap-northeast-2:ACCOUNT_ID:function:notification",
      "Parameters": {
        "proposal_id.$": "$.proposal_id",
        "analysis_result.$": "$.analysis_result",
        "topic_arn": "arn:aws:sns:ap-northeast-2:ACCOUNT_ID:investment-approvals",
        "timestamp.$": "$$.State.EnteredTime"
      },
      "End": true
    },
    "SendRejectionNotification": {
      "Type": "Task",
      "Resource": "arn:aws:lambda:ap-northeast-2:ACCOUNT_ID:function:notification",
      "Parameters": {
        "proposal_id.$": "$.proposal_id",
        "analysis_result.$": "$.analysis_result",
        "topic_arn": "arn:aws:sns:ap-northeast-2:ACCOUNT_ID:investment-rejections",
        "timestamp.$": "$$.State.EnteredTime"
      },
      "End": true
    },
    "HandleError": {
      "Type": "Fail",
      "Cause": "Investment proposal review failed",
      "Error": "ReviewError"
    }
  }
}
EOF

# ACCOUNT_ID 치환
ACCOUNT_ID=$(aws sts get-caller-identity --query Account --output text)
sed -i "s/ACCOUNT_ID/$ACCOUNT_ID/g" state-machine-definition.json

# 상태 머신 생성
aws stepfunctions create-state-machine \
  --name "InvestmentProposalReview" \
  --definition file://state-machine-definition.json \
  --role-arn "arn:aws:iam::$(aws sts get-caller-identity --query Account --output text):role/StepFunctionsExecutionRole"
```
